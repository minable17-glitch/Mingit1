/**
 * 🌟 칭찬팡 도장 - Google Apps Script 웹앱
 * 시트 구성: 학생명단 / 도장기록 / 설정
 * 사용 순서: ① 메뉴 [🌟 칭찬팡 > 초기 설정] 실행 → ② 학생명단 입력 → ③ 웹 앱으로 배포
 */
const SHEET_STUDENTS = '학생명단';
const SHEET_LOG = '도장기록';
const SHEET_SETTINGS = '설정';
const TZ = 'Asia/Seoul';

/* ---------- 웹앱 진입 ---------- */
function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('칭찬팡 도장')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/* ---------- 시트 메뉴 ---------- */
function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('🌟 칭찬팡')
    .addItem('초기 설정 (시트 만들기)', 'setup')
    .addToUi();
}

/* ---------- 초기 설정: 필요한 시트 자동 생성 ---------- */
function setup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let st = ss.getSheetByName(SHEET_STUDENTS);
  if (!st) {
    st = ss.insertSheet(SHEET_STUDENTS);
    st.getRange(1, 1, 1, 3).setValues([['반', '번호', '이름']]).setFontWeight('bold').setBackground('#fff3c4');
    const sample = [];
    for (let i = 1; i <= 5; i++) sample.push(['1', i, '학생' + i]);
    for (let i = 1; i <= 5; i++) sample.push(['2', i, '학생' + (i + 5)]);
    st.getRange(2, 1, sample.length, 3).setValues(sample);
    st.getRange('A:A').setNumberFormat('@'); // 반은 글자로 저장
    st.setFrozenRows(1);
  }
  let lg = ss.getSheetByName(SHEET_LOG);
  if (!lg) {
    lg = ss.insertSheet(SHEET_LOG);
    lg.getRange(1, 1, 1, 5).setValues([['일시', '반', '번호', '이름', '칭찬내용']]).setFontWeight('bold').setBackground('#ffd6d6');
    lg.setFrozenRows(1);
  }
  let se = ss.getSheetByName(SHEET_SETTINGS);
  if (!se) {
    se = ss.insertSheet(SHEET_SETTINGS);
    se.getRange(1, 1, 4, 2).setValues([
      ['항목', '값'],
      ['목표도장수', 10],
      ['칭찬항목', '수업 참여 최고, 친구를 도와줌, 바른 태도, 과제 완벽, 정리정돈, 배려왕, 끝까지 노력'],
      ['도장문구', '참 잘했어요']
    ]);
    se.getRange(1, 1, 1, 2).setFontWeight('bold').setBackground('#d6ecff');
    se.setColumnWidth(2, 500);
  }
  SpreadsheetApp.getActive().toast('칭찬팡 시트가 준비됐어요! 학생명단을 입력해 주세요.', '🌟 칭찬팡');
}

/* ---------- 내부 함수 ---------- */
function key_(cls, num) {
  return String(cls).trim() + '|' + String(num).trim();
}
function sheet_(name) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sh) throw new Error('"' + name + '" 시트가 없어요. 메뉴에서 [초기 설정]을 먼저 실행해 주세요.');
  return sh;
}
function readStudents_() {
  const sh = sheet_(SHEET_STUDENTS);
  const last = sh.getLastRow();
  if (last < 2) return [];
  return sh.getRange(2, 1, last - 1, 3).getDisplayValues()
    .filter(r => r[0] !== '' && r[2] !== '')
    .map(r => ({ cls: String(r[0]).trim(), num: String(r[1]).trim(), name: String(r[2]).trim() }));
}
function readSettings_() {
  const sh = sheet_(SHEET_SETTINGS);
  const map = {};
  const last = sh.getLastRow();
  if (last >= 2) {
    sh.getRange(2, 1, last - 1, 2).getDisplayValues().forEach(r => { map[String(r[0]).trim()] = r[1]; });
  }
  const goal = parseInt(map['목표도장수'], 10);
  return {
    goal: goal > 0 ? goal : 10,
    reasons: String(map['칭찬항목'] || '참 잘했어요').split(',').map(s => s.trim()).filter(Boolean),
    stampText: String(map['도장문구'] || '참 잘했어요').trim()
  };
}
function computeCounts_() {
  const sh = sheet_(SHEET_LOG);
  const last = sh.getLastRow();
  const counts = {};
  if (last < 2) return counts;
  sh.getRange(2, 2, last - 1, 2).getDisplayValues().forEach(r => {
    const k = key_(r[0], r[1]);
    counts[k] = (counts[k] || 0) + 1;
  });
  return counts;
}

/* ---------- 웹앱에서 호출하는 함수 ---------- */
function getInitData() {
  return {
    students: readStudents_(),
    counts: computeCounts_(),
    settings: readSettings_()
  };
}

/** 여러 학생(1명도 가능)에게 도장 찍기 */
function addStamps(list, reason) {
  if (!list || !list.length) throw new Error('학생을 선택해 주세요.');
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sh = sheet_(SHEET_LOG);
    const now = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm:ss');
    const rows = list.map(s => [now, String(s.cls), String(s.num), String(s.name), String(reason || '칭찬')]);
    const start = sh.getLastRow() + 1;
    sh.getRange(start, 1, rows.length, 5).setNumberFormat('@').setValues(rows);
    SpreadsheetApp.flush();
    return computeCounts_();
  } finally {
    lock.releaseLock();
  }
}

/** 해당 학생의 가장 최근 도장 1개 취소 */
function undoLast(cls, num) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sh = sheet_(SHEET_LOG);
    const last = sh.getLastRow();
    if (last >= 2) {
      const vals = sh.getRange(2, 2, last - 1, 2).getDisplayValues();
      const target = key_(cls, num);
      for (let i = vals.length - 1; i >= 0; i--) {
        if (key_(vals[i][0], vals[i][1]) === target) {
          sh.deleteRow(i + 2);
          break;
        }
      }
    }
    SpreadsheetApp.flush();
    return computeCounts_();
  } finally {
    lock.releaseLock();
  }
}

/** 학생 한 명의 최근 칭찬 기록 (최신순 20개) */
function getHistory(cls, num) {
  const sh = sheet_(SHEET_LOG);
  const last = sh.getLastRow();
  if (last < 2) return [];
  const target = key_(cls, num);
  return sh.getRange(2, 1, last - 1, 5).getDisplayValues()
    .filter(r => key_(r[1], r[2]) === target)
    .map(r => ({ time: r[0], reason: r[4] }))
    .reverse()
    .slice(0, 20);
}
