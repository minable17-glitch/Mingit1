<script>
const S = {
  students: [], counts: {}, settings: {goal:10, reasons:[], stampText:'참 잘했어요'},
  cls: null, multi: false, selected: new Set(), sort: 'num',
  target: [], reason: null, busy: false
};
const $ = id => document.getElementById(id);
const key = (c, n) => String(c).trim() + '|' + String(n).trim();
const esc = s => String(s).replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));
const cnt = s => S.counts[key(s.cls, s.num)] || 0;
function toast(msg){ const t=$('toast'); t.textContent=msg; t.classList.add('show'); clearTimeout(t._h); t._h=setTimeout(()=>t.classList.remove('show'),2000); }
function fail(e){ S.busy=false; setBusy(false); toast('⚠️ ' + (e && e.message ? e.message : e)); }
function setBusy(b){ $('stampBtn').disabled=b; $('bulkGo').disabled=b || S.selected.size===0; }
/* ---------- 시작 ---------- */
function load(){
  google.script.run.withSuccessHandler(d => {
    S.students = d.students; S.counts = d.counts; S.settings = d.settings;
    const classes = [...new Set(d.students.map(s => s.cls))];
    S.cls = S.cls && classes.includes(S.cls) ? S.cls : classes[0];
    $('loading').classList.add('hidden');
    renderTabs(); render();
  }).withFailureHandler(e => { $('loading').textContent = '⚠️ ' + (e.message || e); }).getInitData();
}
function classList(){ return [...new Set(S.students.map(s => s.cls))]; }
function renderTabs(){
  $('tabs').innerHTML = classList().map(c =>
    `<button class="tab ${c===S.cls?'on':''}" data-c="${esc(c)}">${esc(c)}반</button>`).join('');
  $('tabs').querySelectorAll('.tab').forEach(b => b.onclick = () => {
    S.cls = b.dataset.c; S.selected.clear(); renderTabs(); render();
  });
}
function current(){
  const list = S.students.filter(s => s.cls === S.cls);
  if (S.sort === 'num') list.sort((a,b) => (parseInt(a.num)||0) - (parseInt(b.num)||0));
  else list.sort((a,b) => cnt(b) - cnt(a) || (parseInt(a.num)||0) - (parseInt(b.num)||0));
  return list;
}
function render(){
  const g = S.settings.goal, list = current();
  const total = list.reduce((a,s) => a + cnt(s), 0);
  $('sum').textContent = list.length ? `${S.cls}반 · ${list.length}명 · 도장 총 ${total}개` : '학생명단 시트에 학생을 입력해 주세요';
  $('grid').innerHTML = list.map(s => {
    const c = cnt(s), filled = c % g, pang = Math.floor(c / g), k = key(s.cls, s.num);
    let dots = ''; for (let i=0;i<g;i++) dots += `<span class="dot ${i<filled?'f':''}"></span>`;
    return `<div class="card ${S.selected.has(k)?'sel':''}" data-k="${esc(k)}">
      ${pang ? `<span class="pangbadge">🎉 팡×${pang}</span>` : ''}
      <span class="check">✓</span>
      <div class="num">${esc(s.num)}번</div>
      <div class="name">${esc(s.name)}</div>
      <div class="dots">${dots}</div>
      <div class="meta"><span>${filled}/${g}</span><span>총 <b>${c}</b>개</span></div>
    </div>`;
  }).join('');
  $('grid').querySelectorAll('.card').forEach(el => el.onclick = () => onCard(el.dataset.k));
  $('bulkBar').classList.toggle('hidden', !S.multi);
  $('bulkInfo').textContent = `${S.selected.size}명 선택`;
  $('bulkGo').disabled = S.selected.size === 0;
}
function findByKey(k){ return S.students.find(s => key(s.cls, s.num) === k); }
function onCard(k){
  if (S.multi){
    S.selected.has(k) ? S.selected.delete(k) : S.selected.add(k);
    render(); return;
  }
  openModal([findByKey(k)]);
}
/* ---------- 모달 ---------- */
function openModal(targets){
  S.target = targets; S.reason = S.settings.reasons[0] || '칭찬';
  $('custom').value = '';
  const single = targets.length === 1;
  if (single){
    const s = targets[0];
    $('mTitle').textContent = `${s.name}`;
    $('mSub').textContent = `${s.cls}반 ${s.num}번 · 지금까지 도장 ${cnt(s)}개`;
    renderBoard(false);
    $('board').classList.remove('hidden');
    $('histWrap').classList.remove('hidden');
    loadHist();
  } else {
    $('mTitle').textContent = `${targets.length}명에게 도장`;
    $('mSub').textContent = targets.map(s => s.name).join(', ');
    $('board').classList.add('hidden');
    $('histWrap').classList.add('hidden');
  }
  renderReasons();
  $('modal').classList.remove('hidden');
}
function closeModal(){ $('modal').classList.add('hidden'); }
function stampMark(){ return `<div class="mark">${esc(S.settings.stampText)}</div>`; }
function renderBoard(animateLast){
  const s = S.target[0], g = S.settings.goal, filled = cnt(s) % g;
  let h = '';
  for (let i=0;i<g;i++){
    const isNew = animateLast && i === filled - 1;
    h += `<div class="slot ${isNew?'new':''}">${i < filled ? stampMark() : (i+1)}</div>`;
  }
  $('board').innerHTML = h;
}
function renderReasons(){
  $('reasons').innerHTML = S.settings.reasons.map(r =>
    `<button class="chip ${r===S.reason?'on':''}" data-r="${esc(r)}">${esc(r)}</button>`).join('');
  $('reasons').querySelectorAll('.chip').forEach(b => b.onclick = () => {
    S.reason = b.dataset.r; $('custom').value=''; renderReasons();
  });
}
function loadHist(){
  const s = S.target[0];
  $('hist').innerHTML = '<li class="empty">불러오는 중…</li>';
  google.script.run.withSuccessHandler(rows => {
    $('hist').innerHTML = rows.length
      ? rows.map(r => `<li><span>${esc(r.reason)}</span><span>${esc(r.time.slice(5,16))}</span></li>`).join('')
      : '<li class="empty">아직 기록이 없어요. 첫 도장을 찍어 주세요!</li>';
  }).withFailureHandler(fail).getHistory(s.cls, s.num);
}
/* ---------- 도장 찍기 ---------- */
function doStamp(){
  if (S.busy) return;
  const reason = $('custom').value.trim() || S.reason || '칭찬';
  const targets = S.target.slice();
  const before = {}; targets.forEach(s => before[key(s.cls,s.num)] = cnt(s));
  S.busy = true; setBusy(true);
  // 먼저 효과 보여주기
  showStampFx(targets.length === 1 ? `${targets[0].name} · ${reason}` : `${targets.length}명 · ${reason}`);
  google.script.run.withSuccessHandler(counts => {
    S.busy = false; setBusy(false);
    S.counts = counts;
    const g = S.settings.goal;
    const done = targets.filter(s => Math.floor(cnt(s)/g) > Math.floor(before[key(s.cls,s.num)]/g));
    if (targets.length === 1){
      const s = targets[0];
      $('mSub').textContent = `${s.cls}반 ${s.num}번 · 지금까지 도장 ${cnt(s)}개`;
      renderBoard(true); loadHist();
    } else {
      closeModal(); S.selected.clear();
    }
    render();
    if (done.length) setTimeout(() => showPang(done), 900);
  }).withFailureHandler(fail).addStamps(targets, reason);
}
function doUndo(){
  const s = S.target[0];
  if (!cnt(s)) { toast('취소할 도장이 없어요'); return; }
  if (!confirm(`${s.name} 학생의 마지막 도장을 취소할까요?`)) return;
  google.script.run.withSuccessHandler(counts => {
    S.counts = counts;
    $('mSub').textContent = `${s.cls}반 ${s.num}번 · 지금까지 도장 ${cnt(s)}개`;
    renderBoard(false); loadHist(); render(); toast('마지막 도장을 취소했어요');
  }).withFailureHandler(fail).undoLast(s.cls, s.num);
}
/* ---------- 효과 ---------- */
function showStampFx(who){
  $('fxMark').textContent = S.settings.stampText;
  $('fxWho').textContent = who;
  const fx = $('stampFx');
  fx.classList.remove('hidden');
  const m = $('fxMark'); m.style.animation='none'; void m.offsetWidth; m.style.animation='';
  if (navigator.vibrate) navigator.vibrate(40);
  clearTimeout(fx._h); fx._h = setTimeout(() => fx.classList.add('hidden'), 850);
}
function showPang(list){
  $('pangMsg').innerHTML = `🎉 ${list.map(s => esc(s.name)).join(', ')}<br>도장판 완성! 칭찬팡 터졌다!`;
  const fx = $('pangFx'); fx.classList.remove('hidden');
  confetti();
  clearTimeout(fx._h); fx._h = setTimeout(() => fx.classList.add('hidden'), 2600);
}
function confetti(){
  const colors = ['#e03131','#ffd43b','#40c057','#228be6','#f783ac','#ff922b'];
  for (let i=0;i<80;i++){
    const p = document.createElement('span');
    p.className = 'confetti';
    p.style.left = Math.random()*100 + 'vw';
    p.style.background = colors[i % colors.length];
    p.style.animationDuration = (1.8 + Math.random()*1.6) + 's';
    p.style.animationDelay = (Math.random()*0.4) + 's';
    p.style.borderRadius = Math.random() > .5 ? '50%' : '2px';
    document.body.appendChild(p);
    setTimeout(() => p.remove(), 4200);
  }
}
/* ---------- 버튼 연결 ---------- */
$('closeBtn').onclick = closeModal;
$('modal').onclick = e => { if (e.target.id === 'modal') closeModal(); };
$('stampBtn').onclick = doStamp;
$('undoBtn').onclick = doUndo;
$('custom').oninput = () => { S.reason = null; renderReasons(); if (!$('custom').value.trim()) { S.reason = S.settings.reasons[0]; renderReasons(); } };
$('multiBtn').onclick = () => {
  S.multi = !S.multi; S.selected.clear();
  $('multiBtn').classList.toggle('on', S.multi);
  $('multiBtn').textContent = S.multi ? '선택 끝내기' : '여러 명 선택';
  render();
};
$('sortBtn').onclick = () => {
  S.sort = S.sort === 'num' ? 'rank' : 'num';
  $('sortBtn').textContent = S.sort === 'num' ? '번호순' : '도장많은순';
  render();
};
$('allBtn').onclick = () => {
  const list = current();
  if (S.selected.size === list.length) S.selected.clear();
  else list.forEach(s => S.selected.add(key(s.cls, s.num)));
  render();
};
$('bulkGo').onclick = () => openModal([...S.selected].map(findByKey).filter(Boolean));
load();
</script>
